cmake_minimum_required(VERSION 3.12)
project(transit_support)
add_library(transit_support SHARED main.cpp module_transit_demod.cpp module_transit_decoder.cpp)
target_link_libraries(transit_support PUBLIC satdump_core)
install(TARGETS transit_support DESTINATION ${CMAKE_INSTALL_LIBDIR}/satdump/plugins)
option(TRANSIT_BUILD_TESTS "Build standalone TRANSIT protocol tests" OFF)
if(TRANSIT_BUILD_TESTS)
  add_executable(transit_protocol_test tests/protocol_test.cpp)
  target_include_directories(transit_protocol_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(transit_product_test tests/product_test.cpp)
  target_include_directories(transit_product_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(transit_product_test PRIVATE satdump_core transit_support)
  add_executable(transit_live_test tests/live_test.cpp)
  target_include_directories(transit_live_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(transit_live_test PRIVATE satdump_core transit_support)
  add_executable(transit_pam_test tests/pam_test.cpp)
  target_include_directories(transit_pam_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(transit_pam_test PRIVATE satdump_core)
endif()
